// Generador aleatorio (Selecciona 5 preguntas únicas usando Fisher-Yates Shuffle)
function generateRandomQuestions(lang) {
    const pool = questionPools[lang] || questionPools['es'];
    const shuffled = [...pool];
    for (let i = shuffled.length - 1; i > 0; i--) {
        const j = Math.floor(Math.random() * (i + 1));
        [shuffled[i], shuffled[j]] = [shuffled[j], shuffled[i]];
    }
    return shuffled.slice(0, 5).map((q, idx) => ({
        ...q,
        text: `${idx + 1}. ${q.text}`
    }));
}

};

// Control de navegación entre pestañas
function switchTab(tab) {
    document.getElementById('sec-quiz').classList.add('hidden');
    document.getElementById('sec-menu').classList.add('hidden');
    document.getElementById('sec-history').classList.add('hidden');

    const defaultBtnClass = "px-4 py-2.5 sm:px-6 sm:py-3 rounded-full text-xs sm:text-sm font-semibold transition-all-custom bg-white/10 backdrop-blur-md text-white hover:bg-white/20 border border-white/20";
    const activeBtnClass = "px-4 py-2.5 sm:px-6 sm:py-3 rounded-full text-xs sm:text-sm font-semibold transition-all-custom bg-olive-600 text-white shadow-lg shadow-olive-900/40 border border-olive-600";

    document.getElementById('tab-quiz').className = defaultBtnClass;
    document.getElementById('tab-menu').className = defaultBtnClass;
    document.getElementById('tab-history').className = defaultBtnClass;

    if (tab === 'quiz') {
        document.getElementById('sec-quiz').classList.remove('hidden');
        document.getElementById('tab-quiz').className = activeBtnClass;
    } else if (tab === 'menu') {
        document.getElementById('sec-menu').classList.remove('hidden');
        document.getElementById('tab-menu').className = activeBtnClass;
    } else if (tab === 'history') {
        document.getElementById('sec-history').classList.remove('hidden');
        document.getElementById('tab-history').className = activeBtnClass;
    }
}

// Renderizado de preguntas del Quiz
function renderQuestion() {
    if (!activeQuestions || activeQuestions.length === 0) return;
    const qData = activeQuestions[currentQuestion];
    document.getElementById('stepNum').innerText = `${currentQuestion + 1} / 5`;
    document.getElementById('progressBar').style.width = `${((currentQuestion + 1) / 5) * 100}%`;
    document.getElementById('questionText').innerText = qData.text;

    const optsContainer = document.getElementById('optionsContainer');
    optsContainer.innerHTML = '';

    qData.options.forEach((opt, idx) => {
        const btn = document.createElement('button');
        btn.className = "w-full text-left p-4 rounded-xl border border-stone-200 hover:border-olive-600 hover:bg-olive-50/50 transition-all flex items-center justify-between group";
        btn.onclick = () => selectOption(opt.profile);
        btn.innerHTML = `
            <span class="text-stone-800 font-medium text-sm sm:text-base group-hover:text-olive-800">${opt.text}</span>
            <span class="w-6 h-6 rounded-full border border-stone-300 flex items-center justify-center group-hover:border-olive-600 group-hover:bg-olive-600 group-hover:text-white text-xs font-bold transition">
                ${String.fromCharCode(65 + idx)}
            </span>
        `;
        optsContainer.appendChild(btn);
    });

    if (currentQuestion > 0) {
        document.getElementById('btnPrev').classList.remove('hidden');
    } else {
        document.getElementById('btnPrev').classList.add('hidden');
    }
}

// Selección de opción y avance
function selectOption(profile) {
    userAnswers[currentQuestion] = profile;
    if (currentQuestion < 4) {
        currentQuestion++;
        renderQuestion();
    } else {
        calculateResult();
    }
}

// Retroceder pregunta
function prevQuestion() {
    if (currentQuestion > 0) {
        currentQuestion--;
        renderQuestion();
    }
}

// Cálculo del vino recomendado y maridaje
function calculateResult() {
    const counts = {};
    userAnswers.forEach(p => counts[p] = (counts[p] || 0) + 1);
    
    let dominantProfile = 'tinto_suave';
    let maxCount = 0;
    for (const p in counts) {
        if (counts[p] > maxCount) {
            maxCount = counts[p];
            dominantProfile = p;
        }
    }

    const matchedWines = wineCatalog.filter(w => w.profile === dominantProfile);
    const wine = matchedWines.length > 0 ? matchedWines[Math.floor(Math.random() * matchedWines.length)] : wineCatalog[0];

    document.getElementById('quiz-container').classList.add('hidden');
    document.getElementById('resultBox').classList.remove('hidden');

    document.getElementById('resWineName').innerText = wine.name;
    document.getElementById('resWineDetails').innerText = `${wine.doc} | ${wine.grapes}`;
    document.getElementById('badgeWineType').innerText = wine.type.toUpperCase();
    document.getElementById('resWineProfile').innerText = wine.tasting;
    document.getElementById('resPairingFood').innerText = wine.pairing;
    document.getElementById('resPairingWhy').innerText = wine.why;
}

// Reinicio del test con preguntas nuevas
function resetQuiz() {
    currentQuestion = 0;
    userAnswers = [];
    activeQuestions = generateRandomQuestions(currentLang);
    document.getElementById('resultBox').classList.add('hidden');
    document.getElementById('quiz-container').classList.remove('hidden');
    renderQuestion();
}

// Renderizado del menú digital
function renderMenu(filter) {
    const grid = document.getElementById('wineGrid');
    grid.innerHTML = '';

    const filtered = filter === 'all' 
        ? wineCatalog 
        : wineCatalog.filter(w => w.type === filter);

    filtered.forEach(wine => {
        const card = document.createElement('div');
        card.className = "bg-white rounded-xl border border-stone-200 p-6 flex flex-col justify-between hover:shadow-lg transition-all border-t-4 border-t-olive-600";
        
        let badgeColor = "bg-winered-600";
        if(wine.type === 'branco') badgeColor = "bg-mustard-500 text-charcoal";
        if(wine.type === 'especial') badgeColor = "bg-olive-700";

        card.innerHTML = `
            <div>
                <div class="flex items-center justify-between mb-3">
                    <span class="text-[10px] font-bold uppercase tracking-wider px-2.5 py-0.5 rounded-full text-white ${badgeColor}">
                        ${wine.doc}
                    </span>
                    <span class="text-xs font-semibold text-stone-600">${wine.year}</span>
                </div>
                <h3 class="serif text-xl font-bold text-charcoal mb-2 leading-snug">${wine.name}</h3>
                <p class="text-xs text-olive-700 font-medium mb-3">🍇 ${wine.grapes}</p>
                <p class="text-stone-600 text-xs leading-relaxed mb-4 line-clamp-4">${wine.tasting}</p>
            </div>

            <div class="pt-4 border-t border-stone-100 space-y-2 text-[11px] text-stone-600">
                <div class="flex justify-between">
                    <span class="font-semibold text-stone-800">Servir:</span>
                    <span>${wine.temp}</span>
                </div>
                <div class="flex justify-between">
                    <span class="font-semibold text-stone-800">Álcool:</span>
                    <span>${wine.alcohol}</span>
                </div>
                <div class="mt-2 pt-2 border-t border-dashed border-stone-200">
                    <span class="font-bold text-winered-600 block mb-0.5">Harmonização:</span>
                    <span class="italic text-stone-600">${wine.pairing}</span>
                </div>
            </div>
        `;
        grid.appendChild(card);
    });
}

// Filtro por categoría de vino
function filterWines(type) {
    ['all', 'tinto', 'branco', 'especial'].forEach(t => {
        const btn = document.getElementById(`filter${t.charAt(0).toUpperCase() + t.slice(1)}`);
        if (t === type) {
            btn.className = "px-3 py-1.5 rounded-full text-xs font-semibold bg-olive-600 text-white";
        } else {
            btn.className = "px-3 py-1.5 rounded-full text-xs font-semibold bg-stone-100 text-stone-700 hover:bg-stone-200";
        }
    });
    renderMenu(type);
}
